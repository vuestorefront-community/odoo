# 💰 Currency Plugin

The `currency` plugin provides a consistent way to format monetary values across the application using the `currency.js` library with configurable settings.

### 🎯 How It Works

The plugin creates a global `$currency` helper that formats numbers into properly localized currency strings based on runtime configuration.

#### Plugin Configuration

```typescript
// 4.currency.ts
import currency from 'currency.js'

export default defineNuxtPlugin(() => {
  const config = useRuntimeConfig()
  
  return {
    provide: {
      currency: (value: number | string) => {
        return currency(value, {
          separator: config.public.currencySeparator,    // Thousands separator
          symbol: config.public.currencySymbol,          // Currency symbol
          decimal: config.public.currencyDecimal,        // Decimal separator
          precision: config.public.currencyPrecision,    // Decimal places
        }).format()
      },
    },
  }
})
```

### 🔧 Configuration

Set up your currency settings in the runtime config:

```typescript
// nuxt.config.ts
export default defineNuxtConfig({
  runtimeConfig: {
    public: {
      currencySymbol: '$',           // Currency symbol (e.g., '$', '€', '£')
      currencySeparator: ',',        // Thousands separator (e.g., ',', '.')
      currencyDecimal: '.',          // Decimal separator (e.g., '.', ',')
      currencyPrecision: 2,          // Number of decimal places
    },
  },
})
```

```bash
# .env
NUXT_PUBLIC_CURRENCY_SYMBOL=$
NUXT_PUBLIC_CURRENCY_SEPARATOR=,
NUXT_PUBLIC_CURRENCY_DECIMAL=.
NUXT_PUBLIC_CURRENCY_PRECISION=2
```

#### In Composables

```typescript
// composables/useProductPricing.ts
export const useProductPricing = () => {
  const { $currency } = useNuxtApp()

  const formatPrice = (price: number) => {
    return $currency(price)
  }

  const formatPriceRange = (minPrice: number, maxPrice: number) => {
    if (minPrice === maxPrice) {
      return formatPrice(minPrice)
    }
    return `${formatPrice(minPrice)} - ${formatPrice(maxPrice)}`
  }

  const calculateDiscount = (originalPrice: number, discountedPrice: number) => {
    const discount = originalPrice - discountedPrice
    const percentage = Math.round((discount / originalPrice) * 100)
    
    return {
      amount: formatPrice(discount),
      percentage: `${percentage}%`,
      savings: `Save ${formatPrice(discount)}`
    }
  }

  return {
    formatPrice,
    formatPriceRange,
    calculateDiscount,
  }
}
```

### 🌍 Multi-Currency Support

Configure different currencies for different locales:

```typescript
// composables/useCurrencyConfig.ts
export const useCurrencyConfig = () => {
  const { locale } = useI18n()
  
  const currencyConfigs = {
    'en-US': {
      symbol: '$',
      separator: ',',
      decimal: '.',
      precision: 2,
    },
    'en-GB': {
      symbol: '£',
      separator: ',',
      decimal: '.',
      precision: 2,
    },
    'de-DE': {
      symbol: '€',
      separator: '.',
      decimal: ',',
      precision: 2,
    },
    'fr-FR': {
      symbol: '€',
      separator: ' ',
      decimal: ',',
      precision: 2,
    },
  }

  const getCurrentCurrencyConfig = () => {
    return currencyConfigs[locale.value] || currencyConfigs['en-US']
  }

  return {
    getCurrentCurrencyConfig,
    currencyConfigs,
  }
}
```

### 🔧 Advanced Configuration

```typescript
// plugins/4.currency.ts - Advanced version
import currency from 'currency.js'

export default defineNuxtPlugin(() => {
  const config = useRuntimeConfig()
  
  return {
    provide: {
      currency: (value: number | string, options?: any) => {
        const defaultOptions = {
          separator: config.public.currencySeparator,
          symbol: config.public.currencySymbol,
          decimal: config.public.currencyDecimal,
          precision: config.public.currencyPrecision,
        }
        
        const mergedOptions = { ...defaultOptions, ...options }
        
        return currency(value, mergedOptions).format()
      },
      
      // Additional helpers
      currencyValue: (value: number | string) => {
        return currency(value, {
          separator: config.public.currencySeparator,
          symbol: '',  // No symbol, just the value
          decimal: config.public.currencyDecimal,
          precision: config.public.currencyPrecision,
        }).format()
      },
    },
  }
})
```

### 🎯 Best Practices

1. **Consistent Formatting**: Always use `$currency()` for monetary values
2. **Proper Configuration**: Set up currency settings in environment variables
3. **Accessibility**: Use proper ARIA labels for screen readers
4. **Responsive Design**: Consider mobile-friendly currency display
5. **Localization**: Support multiple currencies for international users
6. **Performance**: Cache currency configurations when possible
