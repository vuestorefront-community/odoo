# 🍞 Toast Plugin

The `toast` plugin provides a centralized notification system using `vue-toastification` with intelligent error filtering and automatic display of API responses.

### 🎯 How It Works

The plugin integrates toast notifications with automatic error handling and filtering to prevent spam and irrelevant notifications from appearing.

#### Plugin Configuration

```typescript
// plugins/3.toast.client.ts
import Toast, { TYPE } from 'vue-toastification'
import 'vue-toastification/dist/index.css'

export default defineNuxtPlugin((nuxtApp) => {
  nuxtApp.vueApp.use(Toast, {
    timeout: 2000,
    filterBeforeCreate: (toast: any) => {
      if (toast.type === TYPE.ERROR) {
        const msg = typeof toast.content === 'string' ? toast.content.trim().toLowerCase() : ''
        if (!msg || msg === '' || msg.includes('not found')) {
          return false
        }
      }
      return toast
    },
  })
})
```

### 🛡️ Intelligent Error Filtering

The plugin includes smart filtering to prevent unwanted notifications:

**Filtered Error Types:**
- ✅ Empty or whitespace-only messages
- ✅ Generic "not found" errors
- ✅ Undefined or null content

**Benefits:**
- Prevents spam notifications
- Improves user experience
- Reduces notification fatigue
- Maintains clean UI

### 🔗 SDK Integration

The toast system is automatically integrated with the SDK for seamless error handling:

```typescript
// plugins/2.sdk.ts
import { useToast } from 'vue-toastification'

export default defineNuxtPlugin(async () => {
  const sdkConfig = {
    odoo: buildModule<OdooModuleType>(OdooModule, {
      apiUrl: `${config.public.middlewareUrl}api/odoo/`,
      ofetch: $fetch.create({
        credentials: 'include',
        onResponseError({ request, response, options }) {
          if (avoidErrorThrowForSomeRequests(options) && response.status === 500) {
            return
          }
          if (
            response.status === 500
            && response._data
            && response._data?.data?.length > 0
          ) {
            const toast = useToast()
            toast.error(response?._data?.message)
          }
        },
      }),
    }),
  }
})
```

### 💡 Usage Examples

#### In Composables

```typescript
// composables/useAuth.ts
import { useToast } from 'vue-toastification'

export const useAuth = () => {
  const toast = useToast()

  const login = async (params: MutationLoginArgs) => {
    try {
      const data = await $sdk().odoo.mutation<MutationLoginArgs, SignInUserResponse>(
        { mutationName: MutationName.LoginMutation }, 
        { ...params }
      )

      userCookie.value = data?.login?.user?.partner
      user.value = data?.login?.user?.partner as Partner
      router.push('/my-account/personal-data')
    }
    catch (error: any) {
      toast.error(error?.data?.message)
    }
  }

  const updatePartner = async (params: MutationCreateUpdatePartnerArgs) => {
    try {
      const data = await $sdk().odoo.mutation<
        MutationCreateUpdatePartnerArgs,
        CreateUpdatePartnerResponse
      >({ mutationName: MutationName.CreateUpdatePartner }, params)

      user.value = data?.createUpdatePartner
      toast.success('Partner updated successfully')
    }
    catch (error: any) {
      toast.error(error?.data?.message)
    }
  }

  const updatePassword = async (params: MutationUpdatePasswordArgs) => {
    try {
      const data = await $sdk().odoo.mutation<
        MutationUpdatePasswordArgs,
        UpdatePasswordResponse
      >({ mutationName: MutationName.UpdatePasswordMutation }, params)

      toast.success('Password updated successfully')
    }
    catch (error: any) {
      toast.error(error?.data?.message)
    }
  }
}
```

#### In Vue Components

```vue
<template>
  <form @submit.prevent="handleSubmit">
    <input v-model="form.email" type="email" />
    <input v-model="form.password" type="password" />
    <button type="submit">Login</button>
  </form>
</template>

<script setup lang="ts">
import { useToast } from 'vue-toastification'

const toast = useToast()
const form = reactive({
  email: '',
  password: ''
})

const handleSubmit = async () => {
  try {
    await login(form)
    toast.success('Login successful!')
  }
  catch (error) {
    toast.error('Login failed. Please check your credentials.')
  }
}
</script>
```

### 🔧 Advanced Configuration

#### Custom Toast Options

```typescript
// Custom duration and position
toast.success('Message', {
  timeout: 5000,
  position: POSITION.TOP_RIGHT,
  hideProgressBar: false,
  closeOnClick: true,
  pauseOnHover: true,
  draggable: true,
})

// Custom styling
toast.error('Error message', {
  timeout: 0, // Never auto-close
  toastClassName: 'custom-error-toast',
  bodyClassName: 'custom-error-body',
})
```

#### Programmatic Toast Control

```typescript
// Store toast reference for later control
const toastId = toast.success('Processing...', {
  timeout: false,
  closeButton: false,
})

// Update toast content
setTimeout(() => {
  toast.update(toastId, {
    content: 'Processing complete!',
    options: {
      type: 'success',
      timeout: 3000,
    },
  })
}, 2000)

// Dismiss specific toast
toast.dismiss(toastId)

// Clear all toasts
toast.clear()
```

### 🎯 Best Practices

1. **Use Appropriate Types**: Match toast type to message context
2. **Keep Messages Concise**: Short, clear messages work best
3. **Avoid Spam**: Use filtering to prevent excessive notifications
4. **Provide Context**: Include relevant information in messages
5. **Handle Errors Gracefully**: Always show user-friendly error messages
6. **Test Filtering**: Ensure important messages aren't filtered out

### 🔧 Configuration Options

```typescript
// Available configuration options
nuxtApp.vueApp.use(Toast, {
  position: POSITION.TOP_RIGHT,           // Toast position
  timeout: 2000,                   // Auto-dismiss time (ms)
  closeOnClick: true,              // Close on click
  pauseOnFocusLoss: true,         // Pause on window focus loss
  pauseOnHover: true,             // Pause on hover
  draggable: true,                // Allow dragging
  draggablePercent: 0.6,          // Drag threshold
  showCloseButtonOnHover: false,  // Show close button on hover
  hideProgressBar: false,         // Hide progress bar
  closeButton: 'button',          // Close button type
  icon: true,                     // Show icons
  rtl: false,                     // Right-to-left support
})
```
