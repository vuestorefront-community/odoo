# 🔧 H3 Nitro Middleware Layer

The boilerplate leverages Nitro's H3 framework to create a powerful middleware layer that handles authentication, caching, request processing, and automatic data synchronization.


### 🔌 Plugin System

Our boilerplate uses Nitro plugins to automatically handle various operations:

#### Cart Management Plugin

```typescript
// server/plugins/manage-cart.ts
export default defineNitroPlugin((nitro) => {
  nitro.hooks.hook('beforeResponse', async (event, { body }) => {
    if (event.method == 'POST') {
      // Automatically update cart cache after mutations
      await cartAddItem(event, body)
      await cartRemoveItem(event, body)
      await cartUpdateItem(event, body)
      await updateAddress(event, body)
      await addAddress(event, body)
      await createUpdatePartner(event, body)
      await applyCoupon(event, body)
      await applyGiftCard(event, body)
      await clearCartAfterPaymentConfirmation(event, body)
    }
  })
})
```

#### Route Cache Plugin

```typescript
// server/plugins/route-cache.ts
export default defineNitroPlugin((nitroApp) => {
  const handlerList: Handler[] = eval('handlers')
  const skipRoutesSet = new Set(routesToSkipCache)

  const enHandler = handlerList.filter((r) => {
    return !skipRoutesSet.has(r.route)
  })

  if (enHandler.length > 0) {
    enHandler.forEach((handler) => {
      const customHandler = cachedEventHandler(
        lazyEventHandler(handler.handler),
        {
          varies: ['user-agent', 'cookie'],
          group: 'pages',
          name: handler.route,
          getKey: (event: H3Event) => {
            const headers = getRequestHeaders(event)
            const userAgent = headers['user-agent']
            const flags = generateFlags(headers, userAgent)

            return flags.isDesktop ? `desktop-${event.path}` : `mobile-${event.path}`
          },
          maxAge: Number(process.env?.NUXT_SWR_CACHE_TIME),
          swr: true,
        },
      )
      nitroApp.router.use(handler.route, customHandler)
    })
  }
})
```

#### Product Enhancement Plugins

```typescript
// server/plugins/manage-product.ts
export default defineNitroPlugin((nitro) => {
  nitro.hooks.hook('beforeResponse', async (event, { body }) => {
    const requestBody = await readBody(event)

    if (event.method == 'POST' && requestBody[0]?.queryName === QueryName.GetProductVariantQuery) {
      const id = (body as any).productVariant?.product?.id
      const stock = await useStorage('stock').getItem(`stock:product-${id}`)

      // Automatically append stock data to product response
      (body as any).productVariant.product.stock = stock?.[websiteId] || 0
    }
  })
})
```

### 🛠️ H3 Event Handlers

The boilerplate uses H3's event handlers for API endpoints:

#### Basic Event Handler

```typescript
// server/api/odoo/query.post.ts
export default defineEventHandler(async (event: H3Event) => {
  const config = useRuntimeConfig(event)
  const body = await readBody(event)
  
  // Get request headers and cookies
  const headers = getRequestHeaders(event)
  const sessionId = getCookie(event, 'session_id')
  
  // Make request to ODOO
  const response = await $fetch.raw(`${config.public.odooBaseUrl}graphql/vsf`, {
    method: 'POST',
    headers: {
      'accept': 'application/json',
      'content-type': 'application/json',
      'REAL-IP': getRequestIP(event) || '',
      'Cookie': `session_id=${sessionId}`,
    },
    body: JSON.stringify({ query: Queries[body?.[0]?.queryName], variables: body?.[1] }),
  })

  return response?._data?.data
})
```

#### Cached Event Handler

```typescript
// Custom cached function for complex caching logic
const customCache = cachedFunction(
  async (event: H3Event) => {
    // Your data fetching logic
    const data = await fetchDataFromOdoo(event)
    return data
  },
  {
    maxAge: Number(process.env?.NUXT_SWR_CACHE_TIME || 3600),
    staleMaxAge: Number(process.env?.NUXT_SWR_CACHE_TIME || 3600),
    name: 'odoo-query-cache',
    getKey: (event: H3Event) => {
      const body = await readBody(event)
      const country = getHeader(event, 'x-country') || 'US'
      const pricelist = getCookie(event, 'pricelist') || 'default'
      
      return `${body.queryName}-${country}-${pricelist}`
    },
    shouldBypassCache: (event: H3Event) => {
      return event.path.startsWith('/my-account')
    },
  }
)
```

### 🌐 HTTP Utilities

H3 provides comprehensive HTTP utilities:

```typescript
export default defineEventHandler(async (event: H3Event) => {
  // Request information
  const method = getMethod(event)           // GET, POST, etc.
  const url = getRequestURL(event)          // Full URL object
  const headers = getRequestHeaders(event)  // All headers
  const query = getQuery(event)            // Query parameters
  const body = await readBody(event)       // Request body
  
  // Client information
  const ip = getClientIP(event)            // Client IP
  const userAgent = getHeader(event, 'user-agent')
  
  // Cookies
  const sessionId = getCookie(event, 'session_id')
  setCookie(event, 'cart-id', 'value', {
    maxAge: 3600,
    httpOnly: true,
    secure: true,
  })
  
  // Response handling
  setHeader(event, 'x-custom-header', 'value')
  setResponseStatus(event, 200)
  
  return { data: 'response' }
})
```

### 🔄 Hook System

The hook system allows for powerful event-driven architecture:

```typescript
// Plugin using hooks
export default defineNitroPlugin((nitro) => {
  // Before response hook
  nitro.hooks.hook('beforeResponse', async (event, { body }) => {
    // Modify response before sending
    if (event.method === 'POST') {
      await processPostRequest(event, body)
    }
  })
  
  // Request hook
  nitro.hooks.hook('request', async (event) => {
    // Process incoming request
    const startTime = Date.now()
    event.context.startTime = startTime
  })
  
  // Error hook
  nitro.hooks.hook('error', async (error, { event }) => {
    // Handle errors
    console.error('Request error:', error)
  })
})
```
